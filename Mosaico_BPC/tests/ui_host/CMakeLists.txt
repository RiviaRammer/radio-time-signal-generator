cmake_minimum_required(VERSION 3.16)
project(bpc_ui_tests C)
set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:")
set(CMAKE_C_STANDARD 11)
get_filename_component(APP_ROOT "../.." ABSOLUTE)
set(LVGL_DIR "${APP_ROOT}/managed_components/lvgl__lvgl")
file(GLOB_RECURSE LVGL_SOURCES "${LVGL_DIR}/src/*.c")
add_library(lvgl_host STATIC ${LVGL_SOURCES})
target_include_directories(lvgl_host PUBLIC "${LVGL_DIR}" "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(lvgl_host PUBLIC LV_CONF_INCLUDE_SIMPLE _CRT_SECURE_NO_WARNINGS)
target_compile_options(lvgl_host PRIVATE /w)
add_executable(ui_smoke ui_smoke.c "${APP_ROOT}/main/clock_settings.c")
target_include_directories(ui_smoke PRIVATE "${APP_ROOT}/main")
target_compile_options(ui_smoke PRIVATE /experimental:c11atomics /utf-8)
target_link_libraries(ui_smoke PRIVATE lvgl_host)

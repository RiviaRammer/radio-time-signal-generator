cmake_minimum_required(VERSION 3.16)
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
set(MOSAICO_ROOT "$ENV{MOSAICO_ROOT}" CACHE PATH "Mosaico checkout")
set(MOSAICO_BSP_DIR "$ENV{MOSAICO_BSP_DIR}" CACHE PATH "esp-mosaico-bsp component directory")
if(NOT MOSAICO_BSP_DIR AND MOSAICO_ROOT)
    set(MOSAICO_BSP_DIR "${MOSAICO_ROOT}/submodule/esp-mosaico-bsp/components/esp-mosaico-bsp")
endif()
if(NOT EXISTS "${MOSAICO_BSP_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set MOSAICO_BSP_DIR to the BSP component, or MOSAICO_ROOT to a Mosaico checkout")
endif()
set(EXTRA_COMPONENT_DIRS "${MOSAICO_BSP_DIR}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(bpc_clock VERSION 0.3.1)
